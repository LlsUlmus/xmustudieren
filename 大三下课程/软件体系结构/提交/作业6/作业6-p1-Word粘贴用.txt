────────────────────────────────────────────────────────────
一、设计说明（可放在源码前，一段话即可）
────────────────────────────────────────────────────────────

本程序采用双重检查锁定（Double-Checked Locking）：首次访问时用 volatile 修饰的静态变量保存唯一实例，保证多线程可见性；同步块内二次判空，避免重复创建。synchronized 与专用锁对象 INIT_LOCK 配合，保证初始化阶段互斥。SingletonTest 使用多线程并发获取实例，通过打印 identityHashCode（十六进制）验证所有线程得到同一对象。


────────────────────────────────────────────────────────────
二、源代码
────────────────────────────────────────────────────────────

【文件：Singleton.java】

/**
 * 线程安全单例（双重检查锁定）。
 * volatile：新建完成后对其他线程可见，并约束与构造相关的重排序。
 * synchronized：同一时刻只有一个线程执行初始化。
 */
public class Singleton {

    /** 惰性初始化时的共享引用，必须由 volatile 修饰。 */
    private static volatile Singleton instance;

    /** 仅用于初始化互斥，避免占用类字面量锁空间。 */
    private static final Object INIT_LOCK = new Object();

    private Singleton() {}

    public static Singleton getInstance() {
        if (instance == null) {
            synchronized (INIT_LOCK) {
                if (instance == null) {
                    instance = new Singleton();
                }
            }
        }
        return instance;
    }
}


【文件：SingletonTest.java】

import java.util.concurrent.CountDownLatch;

/**
 * 并发拉取单例：若 DCL 正确，所有线程打印的 identity hash 应一致。
 */
public class SingletonTest {

    private static final int THREAD_COUNT = 8;

    public static void main(String[] args) throws InterruptedException {
        CountDownLatch ready = new CountDownLatch(THREAD_COUNT);
        CountDownLatch startGate = new CountDownLatch(1);
        Thread[] workers = new Thread[THREAD_COUNT];

        for (int i = 0; i < THREAD_COUNT; i++) {
            final int index = i + 1;
            workers[i] = new Thread(() -> {
                ready.countDown();
                try {
                    startGate.await();
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                    return;
                }
                Singleton one = Singleton.getInstance();
                System.out.printf("[%s] hash=%s%n",
                        Thread.currentThread().getName(),
                        Integer.toHexString(System.identityHashCode(one)));
            }, "T-" + index);
        }

        for (Thread t : workers) {
            t.start();
        }
        ready.await();
        startGate.countDown();

        for (Thread t : workers) {
            t.join();
        }
    }
}
